cmake_minimum_required(VERSION 3.22)
project(enterprise_gguf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp" CACHE PATH "llama.cpp checkout")
if(NOT EXISTS "${LLAMA_CPP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Missing llama.cpp checkout at ${LLAMA_CPP_DIR}; see ENTERPRISE_PHASE4_README.md")
endif()
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama")
add_library(enterprise_gguf SHARED gguf_jni.cpp)
target_include_directories(enterprise_gguf PRIVATE "${LLAMA_CPP_DIR}/include")
target_link_libraries(enterprise_gguf PRIVATE llama android log)
